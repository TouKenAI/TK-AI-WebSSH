### 1.2 技术选型依据

| 技术 | 选型理由 | 替代方案 | 为什么不选替代 |
|------|---------|---------|---------------|
| Flask | 基底项目在用，简单，适合小团队 | FastAPI、Django | FastAPI异步复杂度高，Django太重 |
| requests | 轻量，直接调HTTP，不用额外SDK | openai SDK | 多一个依赖，且OpenAI SDK对阶跃星辰支持不完全 |
| SQLite | 纯内网单机部署，轻量，无需额外服务 | MySQL、PostgreSQL | 需要额外安装和运维 |
| paramiko | 基底项目在用，成熟稳定 | netmiko | netmiko更适合网络设备，但Phase 1只做Linux |
| xterm.js | 基底项目在用，成熟的Web终端 | 自研 | 自研成本高，xterm.js够用 |

### 2.1 AI代理模块（ai/）

#### 2.1.1 StepFun Provider（ai/stepfun_provider.py）

**职责**：调用阶跃星辰API，生成命令或分析结果

**关键代码**：

```python
# ai/stepfun_provider.py

import requests
import json
import logging

logger = logging.getLogger(__name__)

# 按用户要求，保留API Key明文（方便测试）
STEPFUN_API_KEY = "your-stepfun-api-key-here"
STEPFUN_API_URL = "https://api.stepfun.com/step_plan/v1"

def generate_command(user_input, device_type="linux", context=None):
    """
    生成SSH命令（Phase 1：返回JSON格式，不用function calling）
    
    Args:
        user_input: 用户自然语言输入
        device_type: 设备类型（linux/huawei_vrp/cisco_ios）
        context: 上下文信息（之前执行的命令和结果）
    
    Returns:
        {
            "command": "df -h",
            "description": "查看磁盘使用情况",
            "risk_level": "safe"
        }
    """
    try:
        # 构建 system prompt
        system_prompt = get_system_prompt(device_type)
        
        # 构建 messages
        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_input}
        ]
        
        if context:
            messages.insert(1, {"role": "assistant", "content": json.dumps(context)})
        
        # 调用阶跃星辰 API
        response = requests.post(
            f"{STEPFUN_API_URL}/chat/completions",
            headers={
                "Content-Type": "application/json",
                "Authorization": f"Bearer {STEPFUN_API_KEY}"
            },
            json={
                "model": "step-3.5-flash-2603",
                "messages": messages,
                "response_format": {"type": "json_object"},  # 要求返回JSON
                "max_tokens": 500
            },
            timeout=30
        )
        
        response.raise_for_status()
        
        # 解析响应
        result = response.json()["choices"][0]["message"]["content"]
        command_info = json.loads(result)
        
        # 验证返回格式
        if "command" not in command_info:
            raise ValueError("AI返回格式错误：缺少command字段")
        
        return command_info
        
    except requests.HTTPError as e:
        logger.error(f"阶跃星辰API返回错误: {e.response.status_code} {e.response.text}")
        return {"error": f"AI API返回错误: {e.response.status_code}"}
    except Exception as e:
        logger.error(f"调用阶跃星辰API时发生未知错误: {str(e)}")
        return {"error": f"发生未知错误: {str(e)}"}

def analyze_result(command, result, device_type="linux"):
    """
    分析命令执行结果（Phase 1）
    
    Args:
        command: 执行的命令
        result: 命令执行结果
        device_type: 设备类型
    
    Returns:
        中文分析结果（字符串）
    """
    try:
        system_prompt = "你是一个系统运维专家，请用简洁的中文分析命令执行结果，指出关键信息和建议。"
        
        user_prompt = f"设备类型：{device_type}\n执行的命令：{command}\n执行结果：\n{result}\n\n请用简洁的中文分析以上结果。"
        
        response = requests.post(
            f"{STEPFUN_API_URL}/chat/completions",
            headers={
                "Content-Type": "application/json",
                "Authorization": f"Bearer {STEPFUN_API_KEY}"
            },
            json={
                "model": "step-3.5-flash-2603",
                "messages": [
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_prompt}
                ],
                "max_tokens": 500
            },
            timeout=30
        )
        
        response.raise_for_status()
        
        analysis = response.json()["choices"][0]["message"]["content"]
        return analysis
        
    except Exception as e:
        logger.error(f"分析命令结果时发生错误: {str(e)}")
        return f"分析结果时发生错误：{str(e)}"

def get_system_prompt(device_type):
    """根据设备类型返回system prompt"""
    prompts = {
        "linux": "你是一个Linux系统运维专家。用户不懂命令行，你需要生成准确的Linux命令，并用JSON格式返回，包含command、description、risk_level三个字段。",
        "huawei_vrp": "你是一个华为VRP系统运维专家。用户不懂命令行，你需要生成准确的华为VRP命令，并用JSON格式返回。",
        "cisco_ios": "你是一个Cisco IOS系统运维专家。用户不懂命令行，你需要生成准确的Cisco IOS命令，并用JSON格式返回。",
        "generic": "你是一个网络设备运维专家。请根据设备类型生成正确的命令，并用JSON格式返回。"
    }
    return prompts.get(device_type, prompts["generic"])
```

#### 2.1.2 命令生成器（ai/command_generator.py）

**职责**：根据AI返回的命令信息，生成可执行命令

**关键代码**：

```python
# ai/command_generator.py

import re

def validate_command(command_info):
    """
    验证AI返回的命令信息
    
    Args:
        command_info: AI返回的JSON对象
    
    Returns:
        (is_valid, error_message)
    """
    if "command" not in command_info:
        return False, "缺少command字段"
    
    command = command_info["command"]
    
    # 检查危险命令
    danger_patterns = [
        r"rm\s+-rf\s+/",
        r"mkfs",
        r"dd\s+if=/dev/zero",
        r":\(\)\{\;\:\;\}",  # fork炸弹
    ]
    
    for pattern in danger_patterns:
        if re.search(pattern, command):
            return False, f"检测到危险命令：{command}"
    
    return True, ""

def get_risk_level(command):
    """
    获取命令风险等级
    
    Args:
        command: 命令字符串
    
    Returns:
        "safe" / "warning" / "danger"
    """
    # 安全命令
    safe_patterns = [
        r"^df\s",
        r"^ls\s",
        r"^cat\s",
        r"^head\s",
        r"^tail\s",
        r"^ps\s",
        r"^top\s",
        r"^free\s",
        r"^uname\s",
        r"^uptime\s",
    ]
    
    # 警告命令
    warn_patterns = [
        r"^iptables\s",
        r"^systemctl\s",
        r"^service\s",
        r"^yum\s",
        r"^apt\s",
        r"^kill\s",
        r"^pkill\s",
    ]
    
    for pattern in safe_patterns:
        if re.search(pattern, command):
            return "safe"
    
    for pattern in warn_patterns:
        if re.search(pattern, command):
            return "warning"
    
    return "danger"
```

### 2.2 审计日志模块（audit/）

#### 2.2.1 日志记录器（audit/logger.py）

**职责**：记录所有操作到数据库和文件

**关键代码**：

```python
# audit/logger.py

import sqlite3
import logging
from datetime import datetime

logger = logging.getLogger(__name__)

def log_operation(user_id, device_ip, command, result, ai_analysis=None):
    """
    记录操作日志
    
    Args:
        user_id: 用户ID
        device_ip: 目标设备IP
        command: 执行的命令
        result: 命令执行结果（前500字符）
        ai_analysis: AI分析结果（可选）
    """
    try:
        conn = sqlite3.connect("data/tkai_ssh.db")
        cursor = conn.cursor()
        
        # 创建审计日志表（如果不存在）
        cursor.execute("""
            CREATE TABLE IF NOT EXISTS audit_log (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id TEXT,
                device_ip TEXT,
                command TEXT,
                result TEXT,
                ai_analysis TEXT,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
            )
        """)
        
        # 插入日志
        cursor.execute("""
            INSERT INTO audit_log (user_id, device_ip, command, result, ai_analysis)
            VALUES (?, ?, ?, ?, ?)
        """, (
            user_id,
            device_ip,
            command,
            result[:500] if result else "",
            ai_analysis[:500] if ai_analysis else ""
        ))
        
        conn.commit()
        conn.close()
        
        # 同时写入文件（不可篡改）
        with open("data/logs/audit.log", "a", encoding="utf-8") as f:
            timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            f.write(f"[{timestamp}] user={user_id} device={device_ip} command={command} result={result[:100]}\n")
        
    except Exception as e:
        logger.error(f"记录审计日志时发生错误: {str(e)}")
```

### 5.2 配置热加载

```python
# utils/config.py

import yaml
import os

_config = None
_config_mtime = 0

def get_config():
    """获取配置（带热加载）"""
    global _config, _config_mtime
    
    config_path = "config/config.yaml"
    mtime = os.path.getmtime(config_path)
    
    if _config is None or mtime > _config_mtime:
        with open(config_path, "r", encoding="utf-8") as f:
            _config = yaml.safe_load(f)
        _config_mtime = mtime
    
    return _config
```
