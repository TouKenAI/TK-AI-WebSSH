```python
# ai/stepfun_provider.py

import requests
import json

# 按用户要求保留API Key明文
STEPFUN_API_KEY = "your-stepfun-api-key-here"
STEPFUN_API_URL = "https://api.stepfun.com/step_plan/v1"

def generate_command(user_input, device_type="linux"):
    """生成SSH命令（返回JSON格式）"""
    response = requests.post(
        f"{STEPFUN_API_URL}/chat/completions",
        headers={
            "Content-Type": "application/json",
            "Authorization": f"Bearer {STEPFUN_API_KEY}"
        },
        json={
            "model": "step-3.5-flash-2603",
            "messages": [
                {"role": "system", "content": "你是一个Linux运维专家。请用JSON格式返回命令，包含command、description、risk_level三个字段。"},
                {"role": "user", "content": user_input}
            ],
            "response_format": {"type": "json_object"}
        }
    )
    return json.loads(response.json()["choices"][0]["message"]["content"])
```

```python
# ai/command_generator.py

import re

def check_danger_command(command):
    """检查是否为危险命令"""
    danger_patterns = [
        r"rm\s+-rf\s+/",
        r"mkfs",
        r"dd\s+if=/dev/zero",
        r":\(\)\{\;\:\;\}",  # fork炸弹
    ]
    
    for pattern in danger_patterns:
        if re.search(pattern, command):
            return True, f"检测到危险命令：{command}"
    
    return False, ""
```

```python
# app.py（后端）

@app.route('/api/system/stats')
def get_system_stats():
    """获取系统状态"""
    session_id = request.args.get('session_id')
    ssh_client = get_ssh_client(session_id)
    
    # 执行命令获取CPU使用率
    stdin, stdout, _ = ssh_client.exec_command("top -bn1 | grep 'Cpu(s)' | awk '{print $2}'")
    cpu_usage = stdout.read().decode('utf-8').strip()
    
    # 执行命令获取内存使用率
    stdin, stdout, _ = ssh_client.exec_command("free -m | grep Mem | awk '{print $3/$2 * 100}'")
    memory_usage = stdout.read().decode('utf-8').strip()
    
    return jsonify({
        "success": True,
        "data": {
            "cpu_usage": float(cpu_usage),
            "memory_usage": float(memory_usage)
        }
    })
```
